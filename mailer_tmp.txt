// La Machine à Biophonie — mailer v3 (Oct 31: card lore, own listening page). Receives one packet from the machine's packets page and emails it from this Google account.
const KEY = '__KEY__';
const SUBJECT = 'Your reading from La Machine à Biophonie';
const EXPLAINER = 'https://zackarooso.github.io/tirage/';
const PLAYER = 'https://zackarooso.github.io/biophonie-machine/listen.html?id=';
const FOLDER = 'La Machine à Biophonie — music';

function doPost(e) {
  try {
    const p = JSON.parse(e.postData.contents);
    if ((e.parameter && e.parameter.k) !== KEY) return out({ ok: false, error: 'wrong key' });
    if (p.ping) return out({ ok: true, left: MailApp.getRemainingDailyQuota() });
    const cards = (p.cards || []).filter(String);
    const text = 'You asked:\n\n    “' + p.question + '”\n\nYour words were weighed, one by one, and became music. The numbers of that music drew three cards:\n\n    ' +
      cards.map(function (c) { const L = lore(c); return L ? L[0] + '\n      ' + L[1] + '\n      ' + L[2] : c; }).join('\n\n    ') + '\n\nThe music of your question is attached.\n\nLa Machine à Biophonie\n' + EXPLAINER;
    const music = p.audio ? Utilities.newBlob(Utilities.base64Decode(p.audio), p.audioMime || 'audio/mp4', 'biophonie_music.' + (p.audioExt || 'm4a')) : null;
    const listen = music ? keep(music, p) : '';   // a copy in Drive, played by the machine's own listening page
    const opt = { to: p.to, subject: SUBJECT, body: text + (listen ? '\n\nListen: ' + listen : ''), name: 'La Machine à Biophonie', attachments: [], htmlBody: html(p, cards, listen) };
    if (p.image) opt.inlineImages = { reading: Utilities.newBlob(Utilities.base64Decode(p.image), 'image/jpeg', 'biophonie_reading.jpg') };
    if (music) opt.attachments.push(music);
    MailApp.sendEmail(opt);
    return out({ ok: true, left: MailApp.getRemainingDailyQuota() });
  } catch (err) { return out({ ok: false, error: String(err) }); }
}

// the listening page asks for one recording by its Drive id; only files in the music folder are ever served
function doGet(e) {
  const cb = String((e.parameter && e.parameter.cb) || '').replace(/[^A-Za-z0-9_$.]/g, '');
  let r = { ok: false };
  try {
    const f = DriveApp.getFileById(String(e.parameter.id || ''));
    const parents = f.getParents(); let inFolder = false;
    while (parents.hasNext()) if (parents.next().getName() === FOLDER) inFolder = true;
    if (inFolder) { const b = f.getBlob(); r = { ok: true, q: f.getDescription() || '', mime: b.getContentType(), b64: Utilities.base64Encode(b.getBytes()) }; }
  } catch (err) {}
  const body = JSON.stringify(r);
  return cb ? ContentService.createTextOutput(cb + '(' + body + ');').setMimeType(ContentService.MimeType.JAVASCRIPT)
            : ContentService.createTextOutput(body).setMimeType(ContentService.MimeType.JSON);
}

// saves the recording in a Drive folder anyone-with-the-link can play; never stops the email if Drive says no
function keep(blob, p) {
  try {
    const it = DriveApp.getFoldersByName(FOLDER);
    const folder = it.hasNext() ? it.next() : DriveApp.createFolder(FOLDER);
    const stamp = Utilities.formatDate(new Date(), 'America/Los_Angeles', 'yyyy-MM-dd HHmm');
    const f = folder.createFile(blob.copyBlob().setName('biophonie ' + stamp + ' — ' + String(p.question || '').slice(0, 40) + '.' + (p.audioExt || 'm4a')));
    try { f.setDescription(String(p.question || '').slice(0, 500)); } catch (_) {}
    return PLAYER + f.getId();
  } catch (err) { return ''; }
}

function esc(s) { return String(s || '').replace(/[&<>"]/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]; }); }
